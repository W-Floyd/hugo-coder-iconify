{{- /* figures: lay out two (or more) {{< image >}} shortcodes side by side. They sit in an equal-width flex row and stay side by side at every width, including on mobile. (Any sizes set on an inner image is overwritten.) Usage: {{< figures >}} {{< image src="media/a.avif" alt="..." >}} {{< image src="media/b.avif" alt="..." >}} {{< /figures >}} */ -}} {{- if not (.Page.Store.Get "figuresCSS") -}} {{- .Page.Store.Set "figuresCSS" true -}} {{- end -}} {{- $inner := printf "%s" .Inner -}} {{- $n := len (findRE `]*data-rimg` $inner) -}} {{- if gt $n 0 -}} {{- $sz := printf "(max-width: 768px) %dvw, %dpx" (div 100 $n) (div 860 $n) -}} {{- $inner = replaceRE `sizes="[^"]*"` (printf `sizes="%s"` $sz) $inner -}} {{- end -}}
